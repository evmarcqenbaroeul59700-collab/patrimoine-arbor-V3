function getTreeRowAsObject_(sheet, treeId) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;
  const validationColumn = getValidationEntrepriseColumn_(sheet);
  const rappelColumns = rappelElagageColonnes_(sheet);
  const rappelTz = SpreadsheetApp.openById(SPREADSHEET_ID).getSpreadsheetTimeZone();
  const values = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues();
  for (let i = 0; i < values.length; i++) {
    const row = values[i];
    if (String(row[1]).trim() === String(treeId).trim()) {
      return {
        ...rappelElagageInfos_(row, rappelColumns, rappelTz, rappelElagageAujourdhui_()),
        _rowIndex: i + 2, id: row[1], lat: row[2], lng: row[3], species: row[4],
        height: row[5], dbh: row[6], secteur: row[7], address: row[8], tags: row[9],
        historiqueInterventions: row[10], comment: row[11], photos: row[12],
        etat: row[13], updatedAt: row[14],
        validationEntreprise: validationColumn ? validationEntrepriseBoolean_(row[validationColumn - 1]) : false
      };
    }
  }
  return null;
}

function diffObjects_(before, after) {
  if (!before) return [{ field: "__NEW__", from: null, to: (after && after.id) || "" }];
  const keys = ["lat", "lng", "species", "height", "dbh", "secteur", "address", "tags", "historiqueInterventions", "comment", "photos", "etat", "validationEntreprise"];
  const changes = [];
  keys.forEach(k => {
    const a = before[k], b = after[k];
    const sa = a === null || a === undefined ? "" : String(a);
    const sb = b === null || b === undefined ? "" : String(b);
    if (sa !== sb) changes.push({ field: k, from: a, to: b });
  });
  return changes;
}

const TOKEN_STORE = PropertiesService.getScriptProperties();
const TOKEN_TTL_MS = 1000 * 60 * 60 * 12;

function createToken_() {
  const token = Utilities.getUuid();
  TOKEN_STORE.setProperty(token, String(Date.now()));
  return token;
}

function setTokenMeta_(token, meta) {
  if (!token || !meta) return;
  TOKEN_STORE.setProperty("meta_" + token, JSON.stringify(meta));
}

function getTokenMeta_(token) {
  if (!token) return null;
  const raw = TOKEN_STORE.getProperty("meta_" + token);
  if (!raw) return null;
  try { return JSON.parse(raw); } catch (e) { return null; }
}

function isValidToken_(token) {
  if (!token) return false;
  const ts = TOKEN_STORE.getProperty(token);
  if (!ts) return false;
  const age = Date.now() - Number(ts);
  if (!Number.isFinite(age) || age > TOKEN_TTL_MS) {
    TOKEN_STORE.deleteProperty(token);
    TOKEN_STORE.deleteProperty("meta_" + token);
    return false;
  }
  return true;
}

/* Validation entreprise : colonne repérée par son en-tête, puis ajoutée
 * à droite des colonnes existantes lorsqu'une écriture est nécessaire. */
function validationEntrepriseBoolean_(value) {
  return value === true || value === 1 || ["true", "1", "vrai", "oui"].includes(String(value || "").trim().toLowerCase());
}

function getValidationEntrepriseColumn_(sheet, createIfMissing = false) {
  const lastColumn = sheet.getLastColumn();
  if (lastColumn > 0) {
    const headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
    const index = headers.findIndex(header => String(header || "").trim().toLowerCase().replace(/[\s_-]+/g, "") === "validationentreprise");
    if (index !== -1) return index + 1;
  }
  if (!createIfMissing) return 0;
  const column = Math.max(lastColumn, 15) + 1;
  if (sheet.getMaxColumns() < column) sheet.insertColumnsAfter(sheet.getMaxColumns(), column - sheet.getMaxColumns());
  sheet.getRange(1, column).setValue("validationEntreprise");
  return column;
}

function canValidateEntreprise_(meta) {
  const role = String((meta && meta.role) || "").trim().toLowerCase();
  const secteur = String((meta && meta.secteur) || "").trim().toLowerCase();
  return role === "admin" || (role === "entreprise" && secteur === "perilhon");
}

function handleValidationEntreprise_(data, meta) {
  if (!canValidateEntreprise_(meta)) throw new Error("VALIDATION_ENTREPRISE_INTERDITE");
  const treeId = String(data.treeId || "").trim();
  if (!treeId) throw new Error("ID_ARBRE_MANQUANT");
  const value = data.validationEntreprise === undefined ? true : validationEntrepriseBoolean_(data.validationEntreprise);
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_PATRIMOINE);
  if (!sheet) throw new Error("SHEET_PATRIMOINE_INTROUVABLE");
  if (sheet.getLastRow() < 2) throw new Error("ARBRE_INTROUVABLE");
  const ids = sheet.getRange(2, 2, sheet.getLastRow() - 1, 1).getValues();
  const index = ids.findIndex(row => String(row[0]).trim() === treeId);
  if (index === -1) throw new Error("ARBRE_INTROUVABLE");
  const column = getValidationEntrepriseColumn_(sheet, true);
  const previous = validationEntrepriseBoolean_(sheet.getRange(index + 2, column).getValue());
  const updatedAt = Date.now();
  sheet.getRange(index + 2, column).setValue(value);
  sheet.getRange(index + 2, 15).setValue(updatedAt);
  SpreadsheetApp.flush();
  logHistory_(meta, "VALIDATION_ENTREPRISE", treeId, { from: previous, to: value });
  return { status: "VALIDATION_ENTREPRISE_UPDATED", treeId: treeId, validationEntreprise: value, updatedAt: updatedAt };
}

function handleHistoryGet_(e) {
  const treeId = String((e && e.parameter && e.parameter.id) || "").trim();
  const limit = Number((e && e.parameter && e.parameter.limit) || 50);
  if (!treeId) return jsonResponse({ ok: false, error: "id manquant" });
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const hist = ss.getSheetByName(SHEET_HISTORIQUE);
  if (!hist) return jsonResponse({ ok: true, history: [] });
  const last = hist.getLastRow();
  if (last < 2) return jsonResponse({ ok: true, history: [] });
  const rows = hist.getRange(2, 1, last - 1, hist.getLastColumn()).getValues();
  const out = [];
  for (let i = rows.length - 1; i >= 0; i--) {
    if (String(rows[i][5]).trim() === treeId) {
      out.push({

function getOrCreateTreeFolder(treeId) {
  const root = DriveApp.getFolderById(DRIVE_FOLDER_ID);
  const folders = root.getFoldersByName(treeId);
  return folders.hasNext() ? folders.next() : root.createFolder(treeId);
}

function uploadPhoto(base64, filename, treeId) {
  if (!base64 || !base64.startsWith("data:")) return null;
  const folder = getOrCreateTreeFolder(treeId);
  const match = base64.match(/^data:(.*);base64,/);
  if (!match) return null;
  const bytes = Utilities.base64Decode(base64.split(",")[1]);
  const blob = Utilities.newBlob(bytes, match[1], filename);
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return { driveId: file.getId(), folderId: folder.getId(), url: file.getUrl(), name: filename, addedAt: Date.now() };
}

function deletePhotoFromDrive(driveId) {
  try {
    if (!driveId) return false;
    DriveApp.getFileById(driveId).setTrashed(true);
    return true;
  } catch (e) {
    Logger.log("Erreur suppression photo Drive: " + e);
    return false;
  }
}

      JSON.stringify(allPhotos), data.etat || "", data.updatedAt || Date.now()
    ];
    let isUpdate = false;
    if (lastRow > 1) {
      const ids = sheet.getRange(2, 2, lastRow - 1, 1).getValues();
      for (let i = 0; i < ids.length; i++) {
        if (String(ids[i][0]).trim() === String(data.id).trim()) {
          sheet.getRange(i + 2, 1, 1, rowData.length).setValues([rowData]);
          sheet.getRange(i + 2, validationColumn).setValue(validationEntreprise);
          rappelElagageEcrire_(sheet, i + 2, rappelUpdate);
          sortArbresSheet_(sheet);
          recolorOneArbreById_(sheet, data.id);
          isUpdate = true;
          break;
        }
      }
    }
    if (!isUpdate) {
      sheet.appendRow(rowData);
      sheet.getRange(sheet.getLastRow(), validationColumn).setValue(validationEntreprise);
      rappelElagageEcrire_(sheet, sheet.getLastRow(), rappelUpdate);
      sortArbresSheet_(sheet);
      recolorOneArbreById_(sheet, data.id);
    }
    syncTravaux_(data);
    SpreadsheetApp.flush();
    const afterObj = {
      id: data.id, lat: data.lat || "", lng: data.lng || "", species: data.species || "",
      height: data.height || "", dbh: data.dbh || "", secteur: data.secteur || "", address: data.address || "",
      tags: (data.tags || []).join(","), historiqueInterventions: data.historiqueInterventions || "",
      comment: data.comment || "", photos: JSON.stringify(allPhotos || []), etat: data.etat || "",
      validationEntreprise: validationEntreprise
    };
    logHistory_(meta, isUpdate ? "UPDATE" : "CREATE", data.id, { changes: diffObjects_(beforeObj, afterObj) });
    return ok({ status: isUpdate ? "UPDATED" : "CREATED", photos: allPhotos, validationEntreprise: validationEntreprise });
  } catch (err) {
    return jsonResponse({ ok: false, error: String(err) });
  }
}

function syncTravaux_(data) {
  const etatArbre = String(data.etat || "").trim();
  const ETATS_TRAVAUX = ["Dangereux (A abattre)", "A remplacer", "A surveiller", "A élaguer (URGENT)", "A élaguer (Moyen)", "A élaguer (Faible)"];
  if (!ETATS_TRAVAUX.includes(etatArbre)) return;
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheetTravaux = ss.getSheetByName(SHEET_TRAVAUX);
  if (!sheetTravaux) return;
  const travauxRow = [
    data.id || "", etatArbre || "", data.secteur || "", data.dateDemande || "", data.natureTravaux || "",
    data.address || "", data.species || "", data.dateDemandeDevis || "", data.devisNumero || "", data.montantDevis || "",
    data.dateExecution || "", data.remarquesTravaux || "", data.numeroBDC || "", data.numeroFacture || ""
  ];
  const lastTravaux = sheetTravaux.getLastRow();
  let foundTravaux = false;
  if (lastTravaux > 1) {
    const idsTravaux = sheetTravaux.getRange(2, 1, lastTravaux - 1, 1).getValues();
    for (let i = 0; i < idsTravaux.length; i++) {
      if (String(idsTravaux[i][0]).trim() === String(data.id).trim()) {
        const rowIndex = i + 2;
        sheetTravaux.getRange(rowIndex, 1, 1, travauxRow.length).setValues([travauxRow]);

}

/* EXPORT PDF ANNUEL — CORRECTION INTÉGRÉE */
function annuelDate_(value, timeZone) {
  if (value === null || value === undefined || value === "") return null;
  if (Object.prototype.toString.call(value) === "[object Date]") {
    if (isNaN(value.getTime())) return null;
    return annuelDate_(Utilities.formatDate(value, timeZone, "yyyy-MM-dd"), timeZone);
  }
  const text = String(value).trim();
  if (/^\d{4}-\d{2}-\d{2}T.*(?:Z|[+-]\d{2}:?\d{2})$/i.test(text)) {
    const timestamp = new Date(text);
    return isNaN(timestamp.getTime()) ? null : annuelDate_(timestamp, timeZone);
  }
  let m = text.match(/^(\d{4})-(\d{1,2})-(\d{1,2})(?:$|\s)/);
  let y, month, day;
  if (m) {
    y = Number(m[1]); month = Number(m[2]); day = Number(m[3]);
  } else {
    m = text.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})(?:$|\s|,)/);
    if (!m) return null;
    day = Number(m[1]); month = Number(m[2]); y = Number(m[3]);
  }
  const check = new Date(Date.UTC(y, month - 1, day));
  if (check.getUTCFullYear() !== y || check.getUTCMonth() !== month - 1 || check.getUTCDate() !== day) return null;
  const mm = String(month).padStart(2, "0");
  const dd = String(day).padStart(2, "0");
  return { year: y, iso: `${y}-${mm}-${dd}`, display: `${dd}/${mm}/${y}` };
}

function annuelDateValidation_(text, timeZone) {
  const m = String(text || "").match(/^[^\d\n]{0,16}(\d{4}-\d{1,2}-\d{1,2}|\d{1,2}\/\d{1,2}\/\d{4})(?=$|[\s\],])/);
  return m ? annuelDate_(m[1], timeZone) : null;
}

function annuelChamps_(text) {
  const fields = {};
  const re = /\b(dateDemande|natureTravaux|dateDemandeDevis|devisNumero|montantDevis|dateExecution|remarquesTravaux|numeroBDC|numeroFacture)\s*=\s*([^|\n]*)/g;
  let m;
  while ((m = re.exec(String(text || ""))) !== null) fields[m[1]] = m[2].trim();
  return fields;
}

function annuelHistorique_(text, timeZone) {
  const entries = [];
  let current = "";
  String(text || "").replace(/\r\n?/g, "\n").split("\n").forEach(line => {
    line = line.trim();
    if (!line) return;
    if (annuelDateValidation_(line, timeZone) && current) {
      entries.push(current);
      current = "";
    }
    current += (current ? "\n" : "") + line;
  });
  if (current) entries.push(current);
  return entries.map(entry => {
    const fields = annuelChamps_(entry);
    const execution = annuelDate_(fields.dateExecution, timeZone);
    const validation = annuelDateValidation_(entry, timeZone);
    return { text: entry, fields: fields, execution: execution, date: execution || validation };
  });
}

function annuelIdentite_(treeId, execution, fields) {
  if (!execution) return null;
  const norm = value => String(value || "").trim().replace(/\s+/g, " ").toLowerCase();
  return JSON.stringify([
    String(treeId).trim(), execution.iso, norm(fields.natureTravaux),
    norm(fields.devisNumero), norm(fields.numeroBDC), norm(fields.numeroFacture)
  ]);
}

function annuelCollecter_(patrimoine, travaux, year, timeZone) {
  const entries = [];
  const archived = new Set();
  const seen = new Set();
  let undated = 0;

  patrimoine.forEach(row => {
    const treeId = String(row[1] || "").trim();
    if (!treeId) return;
    annuelHistorique_(row[10], timeZone).forEach(entry => {
      if (!entry.date) { undated++; return; }
      if (entry.date.year !== year) return;
      const exact = JSON.stringify([treeId, entry.text]);
      if (seen.has(exact)) return;
      seen.add(exact);
      const identity = annuelIdentite_(treeId, entry.execution, entry.fields);
      if (identity) archived.add(identity);
      entries.push({
        date: entry.date.iso,
        source: "historique",
        cells: [treeId, row[4] || "", row[7] || "", row[8] || "", year, entry.text]
      });
    });
  });

  travaux.forEach(row => {
    const treeId = String(row[0] || "").trim();
    const execution = annuelDate_(row[10], timeZone);
    // Une demande ou un devis sans date d'exécution ne constitue pas
    // à lui seul une intervention exécutée pendant l'année demandée.
    if (!treeId || !execution || execution.year !== year) return;
    const fields = {
      dateDemande: row[3], natureTravaux: row[4], dateDemandeDevis: row[7],
      devisNumero: row[8], montantDevis: row[9], dateExecution: execution.iso,
      remarquesTravaux: row[11], numeroBDC: row[12], numeroFacture: row[13]
    };
    const identity = annuelIdentite_(treeId, execution, fields);
    if (archived.has(identity) || seen.has(identity)) return;
    seen.add(identity);
    const lines = [
      "Travaux saisis dans le tableau de suivi",
      "Date d’exécution : " + execution.display
    ];
    [
      ["Nature des travaux", fields.natureTravaux], ["Devis n°", fields.devisNumero],
      ["Montant du devis (€)", fields.montantDevis], ["N° BDC", fields.numeroBDC],
      ["N° facture", fields.numeroFacture], ["Remarques", fields.remarquesTravaux]
    ].forEach(pair => {
      if (pair[1] !== "" && pair[1] !== null && pair[1] !== undefined) lines.push(pair[0] + " : " + pair[1]);
    });
    entries.push({
      date: execution.iso,
      source: "travaux",
      cells: [treeId, row[6] || "", row[2] || "", row[5] || "", year, lines.join("\n")]
    });
  });

  entries.sort((a, b) => a.date.localeCompare(b.date) || String(a.cells[0]).localeCompare(String(b.cells[0])));
  return {
    rows: entries.map(entry => entry.cells),
    count: entries.length,
    historiqueCount: entries.filter(entry => entry.source === "historique").length,
    travauxCount: entries.filter(entry => entry.source === "travaux").length,
    undatedHistoryCount: undated
  };
}

function annuelLignesPDF_(rows) {
  const output = [];
  rows.forEach(row => {
    const lines = [];
    String(row[5]).replace(/\r\n?/g, "\n").split("\n").forEach(part => {
      while (part.length > 50) {
        let cut = part.lastIndexOf(" ", 50);
        if (cut < 20) cut = 50;
        lines.push(part.slice(0, cut));
        part = part.slice(cut).replace(/^ /, "");
      }
      lines.push(part);
    });
    // Plusieurs lignes de tableau pour une description longue : les cellules
    // restent lisibles et la suite ne disparaît pas au bord de la page PDF.
    for (let i = 0; i < lines.length; i += 12) {
      output.push(row.slice(0, 5).concat((i ? "(suite)\n" : "") + lines.slice(i, i + 12).join("\n")));
    }
  });
  return output;
}

  const result = annuelCollecter_(readRows(source), readRows(ss.getSheetByName(SHEET_TRAVAUX)), y, timeZone);
  const pdfRows = annuelLignesPDF_(result.rows);

  let tmp = null;
  try {
    // Un nom par export évite qu'une demande supprime le document d'une autre.
    tmp = ss.insertSheet("TMP_EXPORT_ANNUEL_" + Utilities.getUuid().replace(/-/g, "").slice(0, 12));
    const tableStart = 56;
    const lastNeededRow = tableStart + Math.max(1, pdfRows.length);
    if (tmp.getMaxRows() < lastNeededRow) tmp.insertRowsAfter(tmp.getMaxRows(), lastNeededRow - tmp.getMaxRows());
    writeCoverPage_(tmp, `                           Historique annuel des travaux – ${y}`, meta);
    [120, 140, 160, 220, 80, 420].forEach((width, i) => tmp.setColumnWidths(i + 1, 1, width));
    tmp.setRowHeights(1, tableStart - 1, 28);
    tmp.getRange("A45:F46").merge()
      .setValue("Année retenue : date d’exécution des travaux ; à défaut, date de validation dans l’historique.\nLes travaux saisis sans date d’exécution ne sont pas comptés comme travaux exécutés.")
      .setFontSize(FONT_META).setWrap(true).setVerticalAlignment("top");
    tmp.setRowHeights(45, 2, 32);
    tmp.getRange("A48:F49").merge()
      .setValue(`Interventions retenues : ${result.count} (${result.historiqueCount} dans l’historique, ${result.travauxCount} dans le suivi des travaux).` +
        (result.undatedHistoryCount ? `\n${result.undatedHistoryCount} entrée(s) d’historique sans date exploitable : année non déterminée.` : ""))
      .setFontSize(FONT_META).setWrap(true).setVerticalAlignment("top");
    tmp.setRowHeights(48, 2, 32);

    tmp.getRange(tableStart, 1, 1, 6)
      .setValues([["ID Arbre", "Espèce", "Secteur", "Adresse", "Année", "Intervention"]])
      .setFontWeight("bold").setFontSize(FONT_BODY).setWrap(true).setVerticalAlignment("middle");
    if (pdfRows.length) {
      tmp.getRange(tableStart + 1, 1, pdfRows.length, 6).setValues(pdfRows)
        .setFontSize(FONT_BODY).setWrap(true).setVerticalAlignment("top");
      pdfRows.forEach((row, i) => {
        tmp.setRowHeight(tableStart + 1 + i, estimateRowHeight_(row[5], 40, 24) + 12);
      });
    } else {
      tmp.getRange(tableStart + 1, 1, 1, 6).merge()
        .setValue(`Aucune intervention enregistrée pour l’année ${y} dans les données consultées.`)
        .setFontSize(FONT_BODY).setWrap(true).setVerticalAlignment("top");
      tmp.setRowHeight(tableStart + 1, 70);
    }
    SpreadsheetApp.flush();

    const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd_HH-mm");
    const fileName = `Historique_Travaux_${y}_${now}.pdf`;
    const url = `https://docs.google.com/spreadsheets/d/${SPREADSHEET_ID}/export` +
      `?format=pdf&gid=${tmp.getSheetId()}&portrait=true&fitw=true` +
      `&top_margin=${PDF_MARGIN_INCH}&bottom_margin=${PDF_MARGIN_INCH}` +
      `&left_margin=${PDF_MARGIN_INCH}&right_margin=${PDF_MARGIN_INCH}` +
      "&sheetnames=false&printtitle=false&fzr=false&gridlines=false&pagenumbers=true";
    const response = UrlFetchApp.fetch(url, {
      headers: { Authorization: "Bearer " + ScriptApp.getOAuthToken() }
    });
    const blob = response.getBlob().setName(fileName);
    if (blob.getContentType() !== "application/pdf") {
      const bytes = blob.getBytes();
      if (bytes.slice(0, 5).map(b => String.fromCharCode(b)).join("") !== "%PDF-") throw new Error("REPONSE_PDF_INVALIDE");
    }
    const file = DriveApp.getFolderById(DRIVE_FOLDER_ID).createFile(blob);
    return {
      ok: true, url: file.getUrl(), fileUrl: file.getUrl(), name: fileName,
      count: result.count, historiqueCount: result.historiqueCount,
      travauxCount: result.travauxCount, undatedHistoryCount: result.undatedHistoryCount
    };
  } finally {
    if (tmp) {
      try { ss.deleteSheet(tmp); } catch (cleanupError) { Logger.log("Nettoyage export annuel : " + cleanupError); }
    }
  }
}

  "A remplacer": "#000000",
  "A surveiller": "#FFE0B2", "Dangereux (A abattre)": "#FFCDD2",

function findRowByIdPatrimoine_(sheet, treeId) {
  const last = sheet.getLastRow();
  if (last < 2) return -1;
  const ids = sheet.getRange(2, 2, last - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) if (String(ids[i][0]).trim() === String(treeId).trim()) return i + 2;
  return -1;
}

function findRowById_(sheet, treeId) {
  const last = sheet.getLastRow();
  if (last < 2) return -1;
  const ids = sheet.getRange(2, 1, last - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) if (String(ids[i][0]).trim() === treeId) return i + 2;
  return -1;
}

function colorRow_(sheet, rowIndex, etat) {
  const color = ETAT_COLORS[etat];
  if (color) sheet.getRange(rowIndex, 1, 1, sheet.getLastColumn()).setBackground(color)
    .setFontColor(etat === "A remplacer" ? "#ffffff" : "#000000");
}

function syncSurveillance_(data) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_SURVEILLANCE);
  if (!sheet) return;
  const id = String(data.id || "").trim(), etat = String(data.etat || "").trim();
  if (!id) return;
  const rowIndex = findRowByIdPatrimoine_(sheet, id);
  if (etat !== "A surveiller") { if (rowIndex !== -1) sheet.deleteRow(rowIndex); return; }
  const row = [
    new Date(), id, data.lat || "", data.lng || "", data.species || "", data.height || "", data.dbh || "",
    data.secteur || "", data.address || "", (data.tags || []).join(","), data.historiqueInterventions || "",
    data.comment || "", JSON.stringify(data.photos || []), etat, Date.now()
  ];
  if (rowIndex === -1) { sheet.appendRow(row); colorRow_(sheet, sheet.getLastRow(), etat); }
  else { sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]); colorRow_(sheet, rowIndex, etat); }
}

function syncTravauxPartiel_(data, sheetName) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName);
  if (!sheet) return;
  const id = String(data.id || "").trim();
  const etat = String(data.etat || "").trim();
  if (!id) return;
  const row = [
    id, etat, data.secteur || "", data.dateDemande || "", data.natureTravaux || "", data.address || "", data.species || "",
    data.dateDemandeDevis || "", data.devisNumero || "", data.montantDevis || "", data.dateExecution || "",
    data.remarquesTravaux || "", data.numeroBDC || "", data.numeroFacture || ""
  ];
  let rowIndex = findRowById_(sheet, id);
  if (rowIndex === -1) { sheet.appendRow(row); rowIndex = sheet.getLastRow(); }
  else {
    const range = sheet.getRange(rowIndex, 1, 1, row.length);
    const merged = range.getValues()[0].map((cell, i) => cell !== "" && cell !== null ? cell : (row[i] || ""));
    range.setValues([merged]);
  }
  colorRow_(sheet, rowIndex, String(sheet.getRange(rowIndex, 2).getValue() || "").trim());
}

function CLEAN_ORPHAN_TREE_FOLDERS() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(SHEET_PATRIMOINE);
  if (!sheet) throw new Error("SHEET_PATRIMOINE_INTROUVABLE");
  const validIds = new Set();
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 2, sheet.getLastRow() - 1, 1).getValues().forEach(r => { if (r[0]) validIds.add(String(r[0]).trim()); });
  }
  Logger.log("🌳 Arbres actifs : " + validIds.size);
  const folders = DriveApp.getFolderById(DRIVE_FOLDER_ID).getFolders();
  let deleted = 0, kept = 0;
  while (folders.hasNext()) {
    const folder = folders.next(), name = String(folder.getName()).trim();
    if (!name || !validIds.has(name)) { folder.setTrashed(true); deleted++; Logger.log("🗑️ Dossier supprimé : " + name); }
    else kept++;
  }
  Logger.log("✅ Nettoyage terminé");
  Logger.log("🗑️ Dossiers supprimés : " + deleted);
  Logger.log("📁 Dossiers conservés : " + kept);
}
/* Rappels d'élagage : dates enregistrées dans le classeur et contrôle quotidien. */
const RAPPEL_ELAGAGE_ANNEES = 5;
const RAPPEL_ELAGAGE_FUSEAU = "Europe/Paris";
const RAPPEL_ELAGAGE_HANDLER = "ENVOYER_RAPPELS_ELAGAGE";

function rappelElagageColonnes_(sheet, create = false) {
  const names = ["dernierElagageDate", "rappelElagageActif", "rappelElagageEnvoyePour"];
  let width = sheet.getLastColumn();
  const headers = width ? sheet.getRange(1, 1, 1, width).getValues()[0] : [];
  const cols = {};
  names.forEach(name => {
    let index = headers.findIndex(h => String(h || "").trim().toLowerCase() === name.toLowerCase());
    if (index === -1 && create) {
      const column = Math.max(width, 15) + 1;
      if (sheet.getMaxColumns() < column) sheet.insertColumnsAfter(sheet.getMaxColumns(), column - sheet.getMaxColumns());
      sheet.getRange(1, column).setValue(name);
      headers[column - 1] = name;
      width = column;
      index = column - 1;
    }
    cols[name] = index === -1 ? 0 : index + 1;
  });
  return cols;
}

function rappelElagageAujourdhui_() {
  return Utilities.formatDate(new Date(), RAPPEL_ELAGAGE_FUSEAU, "yyyy-MM-dd");
}

function rappelElagageNature_(value) {
  const text = String(value || "").normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase();
  return /\b(elagage|elaguer|elague|elaguee|taille|tailler|taillee)s?\b/.test(text) && !/\b(abattage|dessouchage)\b/.test(text);
}

function rappelElagageDernierHistorique_(history, timeZone, today, etatValidation = "") {
  const entries = annuelHistorique_(history, timeZone);
  let last = "";
  entries.forEach((entry, i) => {
    const nature = entry.fields.natureTravaux || "";
    const pruning = rappelElagageNature_(nature) || (!nature && i === entries.length - 1 && String(etatValidation).startsWith("A élaguer"));
    if (pruning && entry.execution && entry.execution.iso <= today && entry.execution.iso > last) last = entry.execution.iso;
  });
  return last;
}

function rappelElagageEcheance_(iso) {
  if (!iso) return "";
  const parts = iso.split("-").map(Number);
  const year = parts[0] + RAPPEL_ELAGAGE_ANNEES;
  const day = Math.min(parts[2], new Date(Date.UTC(year, parts[1], 0)).getUTCDate());
  return `${year}-${String(parts[1]).padStart(2, "0")}-${String(day).padStart(2, "0")}`;
}

function rappelElagageInfos_(row, cols, timeZone, today) {
  const rawDate = cols.dernierElagageDate ? row[cols.dernierElagageDate - 1] : "";
  const stored = annuelDate_(rawDate, timeZone);
  const last = stored ? stored.iso : rappelElagageDernierHistorique_(row[10], timeZone, today);
  const rawActive = cols.rappelElagageActif ? row[cols.rappelElagageActif - 1] : "";
  const configured = rawActive !== "" && rawActive !== undefined && rawActive !== null;
  const active = configured ? validationEntrepriseBoolean_(rawActive) : !!last;
  const next = rappelElagageEcheance_(last);
  const etat = String(row[13] || "").trim();
  const suspended = etat === "A remplacer" || etat === "Dangereux (A abattre)";
  const sent = cols.rappelElagageEnvoyePour ? annuelDate_(row[cols.rappelElagageEnvoyePour - 1], timeZone) : null;
  return {
    dernierElagageDate: last, prochainElagageDate: next, rappelElagageActif: active,
    rappelElagageConfigure: configured,
    rappelElagageStatut: !active ? "inactif" : suspended ? "suspendu" : !last ? "date_manquante" : next <= today ? "a_programmer" : "a_venir",
    rappelElagageEnvoyePour: sent ? sent.iso : ""
  };
}

function rappelElagagePreparer_(sheet, before, data, meta) {
  if (!meta || String(meta.role || "").toLowerCase() !== "admin") return null;
  const tz = SpreadsheetApp.openById(SPREADSHEET_ID).getSpreadsheetTimeZone();
  const today = rappelElagageAujourdhui_();
  let last = before ? before.dernierElagageDate : "";
  let active = before ? before.rappelElagageActif : false;
  let changed = false;
  const configRequested = validationEntrepriseBoolean_(data.rappelElagageConfigModifiee) ||
    (!before && Object.prototype.hasOwnProperty.call(data, "dernierElagageDate"));
  if (configRequested) {
    const date = annuelDate_(data.dernierElagageDate, tz);
    active = validationEntrepriseBoolean_(data.rappelElagageActif);
    if (active && !date) throw new Error("DATE_DERNIER_ELAGAGE_REQUISE");
    if (date && date.iso > today) throw new Error("DATE_DERNIER_ELAGAGE_FUTURE");
    last = date ? date.iso : "";
    changed = true;
  }
  const previousEtat = before ? before.etat : data.etat;
  if (validationEntrepriseBoolean_(data.validationEntrepriseResetRequested) &&
      previousEtat !== "Dangereux (A abattre)" && previousEtat !== "A remplacer" && data.etat !== "A remplacer") {
    const performed = rappelElagageDernierHistorique_(data.historiqueInterventions, tz, today, previousEtat);
    if (performed && performed > last) {
      last = performed;
      if ((!before || !before.rappelElagageConfigure) && !configRequested) active = true;
      changed = true;
    }
  }
  return changed ? { dernierElagageDate: last, rappelElagageActif: active } : null;
}

function rappelElagageEcrire_(sheet, rowIndex, update) {
  if (!update) return;
  const cols = rappelElagageColonnes_(sheet, true);
  sheet.getRange(rowIndex, cols.dernierElagageDate).setValue(update.dernierElagageDate || "");
  sheet.getRange(rowIndex, cols.rappelElagageActif).setValue(update.rappelElagageActif === true);
}

function rappelElagageEmailValide_(email) {
  return /^[^\s@,;]+@[^\s@,;]+\.[^\s@,;]+$/.test(String(email || "").trim());
}

function getRappelsElagageConfig_(meta) {
  assertAdmin_(meta);
  let installed = false;
  try { installed = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === RAPPEL_ELAGAGE_HANDLER); } catch (e) {}
  return {
    email: TOKEN_STORE.getProperty("ELAGAGE_EMAIL_DESTINATAIRE") || "",
    emailActif: TOKEN_STORE.getProperty("ELAGAGE_EMAIL_ACTIF") === "true",
    declencheurInstalle: installed
  };
}

function saveRappelsElagageConfig_(data, meta) {
  assertAdmin_(meta);
  const email = String(data.email || "").trim();
  const enabled = validationEntrepriseBoolean_(data.emailActif);
  if ((enabled || email) && !rappelElagageEmailValide_(email)) throw new Error("ADRESSE_EMAIL_INVALIDE");
  TOKEN_STORE.setProperty("ELAGAGE_EMAIL_DESTINATAIRE", email);
  TOKEN_STORE.setProperty("ELAGAGE_EMAIL_ACTIF", String(enabled));
  return getRappelsElagageConfig_(meta);
}

  if (!existing) {

  }
  Logger.log("Rappels d’élagage installés : contrôle quotidien le matin, heure de Paris. Configurez l’adresse et activez les e-mails dans l’application.");
}

function ENVOYER_RAPPELS_ELAGAGE() {
  if (TOKEN_STORE.getProperty("ELAGAGE_EMAIL_ACTIF") !== "true") return;
  const recipient = TOKEN_STORE.getProperty("ELAGAGE_EMAIL_DESTINATAIRE") || "";
  if (!rappelElagageEmailValide_(recipient)) return;
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) return;
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName(SHEET_PATRIMOINE);
    if (!sheet || sheet.getLastRow() < 2) return;
    const tz = ss.getSpreadsheetTimeZone();
    const today = rappelElagageAujourdhui_();
    const cols = rappelElagageColonnes_(sheet);
    const rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).getValues();
    const due = [];
    rows.forEach(row => {
      const info = rappelElagageInfos_(row, cols, tz, today);
      if (row[1] && info.rappelElagageStatut === "a_programmer" && info.rappelElagageEnvoyePour !== info.prochainElagageDate) {
        due.push({ id: String(row[1]).trim(), species: row[4] || "", secteur: row[7] || "", address: row[8] || "", next: info.prochainElagageDate });
      }
    });
    if (!due.length || MailApp.getRemainingDailyQuota() < 1) return;
    const display = iso => iso.split("-").reverse().join("/");
    const body = [
      "Rappel d’élagage — patrimoine arboré",
      "Les arbres suivants ont atteint leur échéance de cinq ans depuis le dernier élagage enregistré.", "",
      ...due.map(t => `Arbre ${t.id} — ${t.species}\nSecteur : ${t.secteur}\nAdresse : ${t.address}\nÉchéance : ${display(t.next)}\nFiche : ${BASE_FICHE_URL}?id=${encodeURIComponent(t.id)}\n`),
      "L’alerte reste visible dans l’application jusqu’à mise à jour. Après un nouvel élagage validé, une nouvelle échéance est calculée à cinq ans."
    ].join("\n");
    MailApp.sendEmail({ to: recipient, subject: `Élagage à programmer — ${due.length} arbre${due.length > 1 ? "s" : ""}`, body: body });
    const writeCols = rappelElagageColonnes_(sheet, true);
    // Retrouver chaque arbre par ID : un tri peut avoir déplacé sa ligne.
    const ids = sheet.getRange(2, 2, sheet.getLastRow() - 1, 1).getValues();
    due.forEach(tree => {
      const index = ids.findIndex(row => String(row[0]).trim() === tree.id);
      if (index !== -1) sheet.getRange(index + 2, writeCols.rappelElagageEnvoyePour).setValue(tree.next);
    });
    SpreadsheetApp.flush();
  } catch (error) {
    Logger.log("Rappels d’élagage : " + error);
    throw error;
  } finally {
    lock.releaseLock();
  }
}
